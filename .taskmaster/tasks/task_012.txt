```python
from pathlib import Path

def generate_cache_key(pdf_files):
    """PDF 파일명 + mtime을 포함한 정확한 캐시 키 생성"""
    cache_parts = []
    for file_path in sorted(pdf_files):
        if os.path.exists(file_path):
            mtime = os.path.getmtime(file_path)
            filename = Path(file_path).name
            cache_parts.append(f"{filename}_{int(mtime)}")
    return "_".join(cache_parts)
```

```python
def safe_extract_query(args):
    """안전한 쿼리 추출: args.input→args.query→첫 값 순으로 시도"""
    try:
        # 첫 번째 시도: args.input
        if hasattr(args, 'input') and args.input:
            return args.input
        # 두 번째 시도: args.query  
        if hasattr(args, 'query') and args.query:
            return args.query
        # 세 번째 시도: 첫 번째 속성 값
        if hasattr(args, '__dict__'):
            for key, value in args.__dict__.items():
                if value and isinstance(value, str):
                    return value
        return "질문을 파싱할 수 없습니다"
    except Exception as e:
        logging.warning(f"인자 파싱 실패: {e}")
        return "질문을 파싱할 수 없습니다"
```
