```python
from langchain_core.pydantic_v1 import BaseModel, Field
from typing import Literal

class RouteResponse(BaseModel):
    next: Literal["researcher", "analyst", "END"] = Field(
        description="다음에 호출할 에이전트 선택"
    )

SUPERVISOR_PROMPT = """
당신은 경제 분석팀의 감독관입니다.
사용자의 질문을 분석하여 적절한 작업 순서를 결정하세요.

일반적인 작업 흐름:
1. researcher: 정보 수집이 필요한 경우
2. analyst: 수집된 정보의 분석이 필요한 경우
3. END: 분석이 완료된 경우

현재 대화 상태를 고려하여 다음 단계를 결정하세요.
"""

def supervisor_node(state: AgentState):
    messages = state['messages']
    
    supervisor_llm = llm.with_structured_output(RouteResponse)
    
    response = supervisor_llm.invoke([
        SystemMessage(content=SUPERVISOR_PROMPT),
        HumanMessage(content=f"현재 메시지들: {[m.content for m in messages[-3:]]}")
    ])
    
    return {"next": response.next}
```
