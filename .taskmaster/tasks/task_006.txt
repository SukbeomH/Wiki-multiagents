```python
ANALYST_PROMPT = """
당신은 경제 분석 전문가입니다.
Researcher가 수집한 정보를 바탕으로 다음과 같은 심층 분석을 수행하세요:

1. 핵심 트렌드 및 패턴 식별
2. 원인과 결과 관계 분석
3. 잠재적 영향 및 시사점 도출
4. 근거 기반 결론 제시

분석 결과는 전문적이면서도 이해하기 쉽게 구조화해주세요.
"""

def analyst_node(state: AgentState):
    messages = state['messages']
    research_info = ""
    
    # 이전 메시지들에서 연구 정보 추출
    for msg in messages:
        if hasattr(msg, 'content'):
            research_info += msg.content + "\n"
    
    analysis_prompt = f"""
    {ANALYST_PROMPT}
    
    연구 정보:
    {research_info}
    
    위 정보를 바탕으로 종합적인 경제 분석 보고서를 작성해주세요.
    
    보고서 구조:
    ## 핵심 요약
    ## 상세 분석
    ## 시사점 및 전망
    ## 결론
    """
    
    response = llm.invoke([SystemMessage(content=analysis_prompt)])
    return {"messages": [response], "next": "END"}
```
