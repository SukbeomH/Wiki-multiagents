import faiss
from langchain.retrievers import ContextualCompressionRetriever
from langchain.retrievers.document_compressors import LLMChainExtractor

def save_faiss_index(vectorstore, index_path):
    vectorstore.save_local(index_path)

def load_faiss_index(index_path):
    return FAISS.load_local(index_path)

def create_mmr_retriever(vectorstore, k=5, fetch_k=20):
    return vectorstore.as_retriever(
        search_type="mmr",
        search_kwargs={"k": k, "fetch_k": fetch_k}
    )

def create_compressed_retriever(base_retriever):
    compressor = LLMChainExtractor.from_llm(llm)

    )

6. 운영 로깅 강화 (logging_config.py):
```python
import logging
import json
from logging.handlers import RotatingFileHandler

def setup_operational_logging():
    logger = logging.getLogger("economic_analysis")
    
    # JSON 라인 포맷터
    class JSONLineFormatter(logging.Formatter):
        def format(self, record):
            log_entry = {
                "timestamp": self.formatTime(record),
                "level": record.levelname,
                "message": record.getMessage(),
                "module": record.module
            }
            return json.dumps(log_entry, ensure_ascii=False)
    
    # 파일 핸들러 설정
    file_handler = RotatingFileHandler(
        "logs/app.log", maxBytes=10*1024*1024, backupCount=5
    )
    file_handler.setFormatter(JSONLineFormatter())
    logger.addHandler(file_handler)

# DDGS 재시도/폴백 로직
def search_with_fallback(query, max_retries=3):
    for attempt in range(max_retries):
        try:
            return ddgs_search(query)
        except Exception as e:
            logger.warning(f"DDGS 검색 실패 (시도 {attempt+1}): {e}")
            if attempt == max_retries - 1:
                return fallback_search(query)
```
